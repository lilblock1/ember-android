cmake_minimum_required(VERSION 3.22.1)
project(andes_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(probe_core STATIC probe_core.cpp)
set_target_properties(probe_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    add_library(andes_probe SHARED bridge.cpp)
    target_link_libraries(andes_probe PRIVATE probe_core GLESv3 android)
    # Windows 8.3 compiler aliases can hide clang++'s name; retain C++ runtime linkage.
    target_link_options(andes_probe PRIVATE --driver-mode=g++)
    target_compile_options(andes_probe PRIVATE -Wall -Wextra -Werror)
else()
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(probe_core_test probe_core_test.cpp)
    target_link_libraries(probe_core_test PRIVATE probe_core Threads::Threads)
    add_test(NAME probe_core_test COMMAND probe_core_test)
endif()
